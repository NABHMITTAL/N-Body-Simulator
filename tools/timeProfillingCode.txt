import universe
import renderer
import time

class Simulation:
  def __init__(self):
    self.universe = universe.Universe()
    self.renderer = renderer.Renderer()
    self.selected_body_index = 0
    self.renderer.camera.follow_target = self.universe.bodies[self.selected_body_index]
    self.max_substeps = 500

    self.selected_body = 0

    self.dt = 0.01
    self.time_scale_index = 0
    self.time_scales = [1,10,100,1000,5000,10000,50000,100000,1000000,10000000]

    # -------- Performance Profiling --------
    self.profile_frames = 0

    self.physics_time = 0.0
    self.render_time = 0.0
    self.camera_time = 0.0
    self.event_time = 0.0
    self.frame_time = 0.0



  def check_events(self):
    #switch body
    if self.renderer.switch_body_requested:
      self.selected_body_index += 1
      if self.selected_body_index >= len(self.universe.bodies):
          self.selected_body_index = 0  
      self.renderer.camera.follow_target = self.universe.bodies[self.selected_body_index]
      self.renderer.switch_body_requested = False

    #change time
    if self.renderer.time_up_requested:
      if self.time_scale_index<9:
        self.time_scale_index+= 1
        self.renderer.time_up_requested = False
    if self.renderer.time_down_requested:
      if self.time_scale_index>0:
        self.time_scale_index-= 1
        self.renderer.time_down_requested = False
    if self.renderer.time_to_one:
        self.time_scale_index = 0
        self.renderer.time_to_one = False

  def run(self):
    while self.renderer.state:
      frame_start = time.perf_counter()

      # ================= Events =================
      start = time.perf_counter()

      self.renderer.process_events()
      self.check_events()

      self.event_time += time.perf_counter() - start
      # ==========================================

      time_warp = self.time_scales[self.time_scale_index]
      substeps = min(time_warp, self.max_substeps)

      # ================= Physics ================
      start = time.perf_counter()

      for _ in range(substeps):
          self.universe.leapFrog(self.dt)

      self.physics_time += time.perf_counter() - start
      # ==========================================

      # ================= Camera =================
      start = time.perf_counter()

      self.renderer.camera.update()

      self.camera_time += time.perf_counter() - start
      # ==========================================

      # ================= Render =================
      start = time.perf_counter()

      self.renderer.draw(self.universe, time_warp)

      self.render_time += time.perf_counter() - start
      # ==========================================

      self.frame_time += time.perf_counter() - frame_start

      self.profile_frames += 1

      if self.profile_frames == 100:
          print(f"Warp Level: {self.time_scales[self.time_scale_index]}")
          print("\n==============================")
          print("Performance Profile (100 frames)")
          print("==============================")

          print(f"Physics : {(self.physics_time/100)*1000:.3f} ms")
          print(f"Rendering : {(self.render_time/100)*1000:.3f} ms")
          print(f"Camera : {(self.camera_time/100)*1000:.3f} ms")
          print(f"Events : {(self.event_time/100)*1000:.3f} ms")
          print(f"Frame : {(self.frame_time/100)*1000:.3f} ms")

          print("==============================\n")

          self.profile_frames = 0

          self.physics_time = 0
          self.render_time = 0
          self.camera_time = 0
          self.event_time = 0
          self.frame_time = 0
